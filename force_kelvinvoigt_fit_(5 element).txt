import pandas as pd
import numpy as np
from scipy.optimize import curve_fit
import matplotlib.pyplot as plt

# 1. Load Data
file_path = "Force_P15A.csv"
df = pd.read_csv(file_path)
df.columns = df.columns.str.strip()

# Conversion: (grams / 1000) * 9.80665
df["Force_N"] = (df["Force"] / 1000) * 9.80665

time = df["Time"].to_numpy()
force = df["Force_N"].to_numpy()

# 2. Robust Detection
abs_max_idx = np.argmax(force)
slope = np.gradient(force, time)
cliff_indices = np.where((time > time[abs_max_idx]) & (slope < -1.96))[0]

if len(cliff_indices) > 0:
    idx_start = cliff_indices[0]
else:
    idx_start = abs_max_idx

t_start = time[idx_start]
t0 = time[idx_start]

time_decay = time[idx_start:]
force_decay = force[idx_start:]
t_fit = time_decay - t0

# 3. Model and Fit
def double_exp(t, A2, tau2, A3, tau3, C):
    return A2 * np.exp(-t / tau2) + A3 * np.exp(-t / tau3) + C

p0 = [force_decay[0] * 0.6, 0.1, force_decay[0] * 0.4, 1.0, force_decay[-1]]
params, cov = curve_fit(double_exp, t_fit, force_decay, p0=p0, maxfev=20000)

# 4. R2 Calculation
y_pred = double_exp(t_fit, *params)
r_squared = 1 - (np.sum((force_decay - y_pred)**2) / np.sum((force_decay - np.mean(force_decay))**2))

# 5. Plotting
plt.figure(figsize=(8, 6))

# SET FONT SIZES EXPLICITLY HERE
fs = 14 # Change this number to make text larger or smaller

plt.plot(time, force, label="Raw data")
plt.plot(time_decay, y_pred, label="Double exp fit")
plt.axvline(t_start, linestyle="--", label="Max force point")

plt.xlabel("Time (s)", fontsize=fs)
plt.ylabel("Force (N)", fontsize=fs)
plt.legend(fontsize=fs-2)
plt.xticks(fontsize=fs-2)
plt.yticks(fontsize=fs-2)

plt.show()

# 6. Display Table
df_params = pd.DataFrame({
    "Parameter": ["A2", "tau2 (s)", "A3", "tau3 (s)", "C", "R-Squared"],
    "Value": np.append(params, r_squared)
})
